-- gmv_daily · 派生指标 · v2.1
INSERT OVERWRITE TABLE dwt_loan_daily_df
PARTITION (ds)
SELECT
    t.ds,
    SUM(t.pay_amt) AS gmv,
    COUNT(DISTINCT t.loan_id) AS order_cnt,
    COUNT(DISTINCT t.reader_id) AS user_cnt
FROM dwd_loan_circ_df t
WHERE t.order_status IN (20, 30, 40)
  AND t.ds BETWEEN '20260701' AND '20260803'
GROUP BY t.ds
;
