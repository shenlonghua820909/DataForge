-- dwd_bib_unified_df · 书目统一明细事实表 · v3.2 · 严格遵循 MARC21 / CNMARC
CREATE TABLE IF NOT EXISTS dwd_bib_unified_df (
    bib_id          STRING          COMMENT '书目记录 ID, MARC21 001',
    isbn            STRING          COMMENT 'ISBN 13位, MARC21 020',
    issn            STRING          COMMENT 'ISSN, MARC21 022',
    title           STRING          COMMENT '正题名, MARC21 245$a',
    subtitle        STRING          COMMENT '副题名, MARC21 245$b',
    author_id       STRING          COMMENT '$主要责任者ID, MARC21 100/110/111',
    author_name     STRING          COMMENT '$主要责任者姓名, MARC21 100$a',
    publisher_id    STRING          COMMENT '$出版社ID, MARC21 264$b',
    publisher_name  STRING          COMMENT '$出版者, MARC21 264$b',
    pub_year        INT             COMMENT '$出版年, MARC21 008/7-10',
    language        STRING          COMMENT '$语种 ISO 639, MARC21 008/35-37',
    clc_code        STRING          COMMENT '$中图分类号, 中图法第五版',
    subject_id      STRING          COMMENT '$主题词ID, MARC21 6XX',
    doc_type        STRING          COMMENT '$文献类型, MARC21 leader/06',
    carrier_type    STRING          COMMENT '$载体类型, MARC21 008/23',
    price           DECIMAL(18,2)   COMMENT '$定价, MARC21 020$c',
    page_count      INT             COMMENT '$页数, MARC21 300$a',
    marc_xml        STRING          COMMENT '$MARC21 ISO 2709 原始记录',
    status          STRING          COMMENT '$书目状态: draft/published/weeded',
    source_system   STRING          COMMENT '$来源系统: 汇文/ALEPH/读秀/CALIS',
    catalog_date    DATE            COMMENT '$编目完成日期',
    gmt_create      TIMESTAMP       COMMENT '$系统入库时间',
    gmt_modified    TIMESTAMP       COMMENT '$最后修改时间',
    ds              STRING          COMMENT '$业务日期 yyyymmdd 分区'
) COMMENT '书目统一明细事实表, CALIS 编目规范'
PARTITIONED BY (ds STRING)
STORED AS ALIORC
TBLPROPERTIES ('life_cycle'='365', 'comment'='MARC21/CNMARC standard');
